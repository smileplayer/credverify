pragma solidity ^0.8.24;

/// @title CredentialRegistry
/// @notice Sổ đăng ký chứng chỉ khóa học của MỘT đơn vị phát hành.
///         Mỗi issuer có một instance riêng.
///
/// State machine của chứng chỉ:  None -> Issued -> Revoked  (một chiều)
///
/// ---------------------------------------------------------------------------
///  BỐN TẦNG QUYỀN
/// ---------------------------------------------------------------------------
///  owner        Đơn vị vận hành (deployer). Hành động BẢO VỆ thì tức thì
///               (setDisputed, removeIssuer); hành động PHÁ HỦY thì phải qua
///               timelock và issuerAdmin phủ quyết được.
///
///  issuerAdmin  Khóa LẠNH của đơn vị phát hành — két sắt, ví phần cứng.
///               Phủ quyết, gỡ tranh chấp, thêm/xóa khóa cấp, chỉ định khóa kế
///               nhiệm, thu hồi kể cả khi đang tranh chấp, và quyền thoát ly.
///
///  isIssuer[]   Khóa NÓNG dùng hằng ngày. Chỉ cấp và thu hồi nghiệp vụ.
///               Bị khóa khi trạng thái là Disputed.
///
///  bất kỳ ai    Thi hành một đề xuất đã qua thời gian chờ mà không bị phủ quyết.
/// ---------------------------------------------------------------------------
///
///  GIẢ ĐỊNH TIN CẬY: khóa issuerAdmin được đơn vị phát hành bảo quản an toàn.
///  Hệ thống KHÔNG phòng vệ được trước việc khóa này bị chiếm — mất nó tương
///  đương mất quyền kiểm soát tổ chức. Đường thoát cuối cùng là di trú sang
///  contract mới và đổi neo tên miền (docs/SECURITY-AUDIT.md mục 2.7).
contract CredentialRegistry {
    // ---------- Kiểu dữ liệu ----------

    enum Status {
        None,     // 0 - chưa từng tồn tại
        Issued,   // 1 - đã cấp, còn hợp lệ
        Revoked   // 2 - đã bị thu hồi, vĩnh viễn không hợp lệ
    }

    enum IssuerStatus {
        Active,   // 0 - hoạt động bình thường
        Disputed  // 1 - đang tranh chấp: khóa nóng bị vô hiệu hóa
    }

    struct Certificate {
        bytes32 certHash;   // hash nội dung tệp chứng chỉ gốc
        address issuer;     // khóa đã cấp
        Status  status;     // trạng thái hiện tại
        uint64  issuedAt;   // thời điểm cấp
        address holder;     // ví học viên
        uint64  revokedAt;  // thời điểm thu hồi (0 nếu chưa thu hồi)
    }

    // ---------- Trạng thái lưu trữ ----------

    address public owner;
    address public pendingOwner;

    address public issuerAdmin;
    address public pendingIssuerAdmin;

    IssuerStatus public issuerStatus;

    mapping(address => bool)    public isIssuer;

    /// @notice Số khóa cấp ĐANG có quyền, ngay lúc này.
    uint256 public activeIssuerCount;
    address[] private _knownIssuers;              // mọi địa chỉ TỪNG được cấp quyền
    mapping(address => bool) private _everListed; // chống trùng khi cấp lại quyền
    mapping(address => address) public issuerSuccessor;   // khóa cũ -> khóa kế nhiệm
    mapping(bytes32 => Certificate) public certificates;

    /// @notice Thời gian chờ trước khi owner thi hành được một đề xuất thu hồi cứu hộ.
    uint64 public immutable gracePeriod;

    /// @notice Thời gian chờ trước khi issuerAdmin hoàn tất việc thoát ly.
    uint64 public immutable exitDelay;

    mapping(bytes32 => uint64) public pendingRevoke;   // certId -> thời điểm được phép thi hành
    uint64 public exitRequestedAt;                     // 0 = chưa yêu cầu thoát ly

    // ---------- Sự kiện ----------

    event IssuerAdded(address indexed issuerAddress);
    event IssuerRemoved(address indexed issuerAddress, address indexed by);
    event IssuerSuccessorSet(address indexed oldIssuer, address indexed newIssuer);

    event CertificateIssued(
        bytes32 indexed certId,
        bytes32 indexed certHash,
        address issuer,
        address indexed holder,
        uint256 issuedAt
    );
    event CertificateRevoked(bytes32 indexed certId, address indexed by, uint256 revokedAt);

    event IssuerDisputed(address indexed by, string reason);
    event IssuerRestored(address indexed by);

    event OwnerRevokeProposed(bytes32 indexed certId, string reason, uint256 executeAfter);
    event OwnerRevokeCancelled(bytes32 indexed certId, address indexed by);

    event OwnershipTransferStarted(address indexed from, address indexed to);
    event OwnershipTransferred(address indexed from, address indexed to);
    event IssuerAdminTransferStarted(address indexed from, address indexed to);
    event IssuerAdminTransferred(address indexed from, address indexed to);

    event SelfSovereigntyRequested(uint256 executeAfter);
    event SelfSovereigntyCancelled();
    event SelfSovereigntyExecuted(address indexed newOwner);

    // ---------- Modifier ----------

    modifier onlyOwner() {
        require(msg.sender == owner, "CredentialRegistry: caller is not owner");
        _;
    }

    modifier onlyIssuerAdmin() {
        require(msg.sender == issuerAdmin, "CredentialRegistry: caller is not issuer admin");
        _;
    }

    modifier onlyIssuer() {
        require(isIssuer[msg.sender], "CredentialRegistry: caller is not an authorized issuer");
        _;
    }

    /// @dev Chặn khóa nóng khi đang tranh chấp. KHÔNG áp cho verifyCertificate,
    ///      và KHÔNG áp cho đường thu hồi của issuerAdmin — phanh không bao giờ bị khóa.
    modifier whenNotDisputed() {
        require(issuerStatus == IssuerStatus.Active, "CredentialRegistry: issuer is disputed");
        _;
    }

    // ---------- Constructor ----------

    /// @param _issuerAdmin      Khóa lạnh của đơn vị phát hành.
    /// @param _firstIssuingKey  Khóa nóng đầu tiên được phép cấp chứng chỉ.
    /// @param _gracePeriod      Thời gian chờ cho đề xuất thu hồi cứu hộ (giây).
    /// @param _exitDelay        Thời gian chờ cho quyền thoát ly (giây), phải > _gracePeriod.
    constructor(
        address _issuerAdmin,
        address _firstIssuingKey,
        uint64  _gracePeriod,
        uint64  _exitDelay
    ) {
        require(_issuerAdmin != address(0), "CredentialRegistry: zero issuer admin");
        require(_firstIssuingKey != address(0), "CredentialRegistry: zero issuing key");
        require(_exitDelay > _gracePeriod, "CredentialRegistry: exitDelay must exceed gracePeriod");

        owner = msg.sender;
        issuerAdmin = _issuerAdmin;
        gracePeriod = _gracePeriod;
        exitDelay = _exitDelay;

        isIssuer[_firstIssuingKey] = true;
        activeIssuerCount = 1;
        _everListed[_firstIssuingKey] = true;
        _knownIssuers.push(_firstIssuingKey);
        emit IssuerAdded(_firstIssuingKey);
    }

    // ---------- Quản lý khóa cấp (khóa nóng) ----------

    function addIssuer(address issuerAddress) external onlyIssuerAdmin {
        require(issuerAddress != address(0), "CredentialRegistry: zero address");
        require(!isIssuer[issuerAddress], "CredentialRegistry: already an issuer");
        isIssuer[issuerAddress] = true;
        activeIssuerCount += 1;
        if (!_everListed[issuerAddress]) {
            _everListed[issuerAddress] = true;
            _knownIssuers.push(issuerAddress);
        }
        emit IssuerAdded(issuerAddress);
    }

    /// @notice Cả owner lẫn issuerAdmin đều gỡ được
    function removeIssuer(address issuerAddress) external {
        require(msg.sender == owner || msg.sender == issuerAdmin,
                "CredentialRegistry: caller is not owner or issuer admin");
        require(isIssuer[issuerAddress], "CredentialRegistry: not an issuer");
        isIssuer[issuerAddress] = false;
        activeIssuerCount -= 1;
        emit IssuerRemoved(issuerAddress, msg.sender);
    }

    /// @notice Chỉ định khóa kế nhiệm để tiếp quản quyền thu hồi các chứng chỉ đã cấp bằng một khóa cũ.
    function setIssuerSuccessor(address oldIssuer, address newIssuer) external onlyIssuerAdmin {
        require(oldIssuer != address(0) && newIssuer != address(0), "CredentialRegistry: zero address");
        issuerSuccessor[oldIssuer] = newIssuer;
        emit IssuerSuccessorSet(oldIssuer, newIssuer);
    }

    // ---------- Trạng thái tranh chấp ----------

    /// @notice Vô hiệu hóa mọi khóa nóng ngay lập tức. Dùng khi nghi ngờ lộ khóa mà chưa xác định được khóa nào.
    function setDisputed(string calldata reason) external {
        require(msg.sender == owner || msg.sender == issuerAdmin,
                "CredentialRegistry: caller is not owner or issuer admin");
        issuerStatus = IssuerStatus.Disputed;
        emit IssuerDisputed(msg.sender, reason);
    }

    /// @notice Gỡ tranh chấp. issuerAdmin gỡ được TỨC THÌ.
    function restore() external {
        require(msg.sender == owner || msg.sender == issuerAdmin,
                "CredentialRegistry: caller is not owner or issuer admin");
        issuerStatus = IssuerStatus.Active;
        emit IssuerRestored(msg.sender);
    }

    // ---------- Luồng nghiệp vụ chính ----------

    function issueCertificate(bytes32 certId, bytes32 certHash, address holder)
        external
        onlyIssuer
        whenNotDisputed
    {
        require(holder != address(0), "CredentialRegistry: holder is zero address");
        // Hash rỗng phải bị từ chối: nếu lọt, verifyCertificate(certId, 0x00..0) sẽ khớp
        // và mọi tệp rỗng đều "hợp lệ". Xem SECURITY-AUDIT SC-05.
        require(certHash != bytes32(0), "CredentialRegistry: empty certHash");
        require(certificates[certId].status == Status.None, "CredentialRegistry: certId already used");

        certificates[certId] = Certificate({
            certHash: certHash,
            issuer: msg.sender,
            status: Status.Issued,
            issuedAt: uint64(block.timestamp),
            holder: holder,
            revokedAt: 0
        });

        emit CertificateIssued(certId, certHash, msg.sender, holder, block.timestamp);
    }

    /// @notice Thu hồi một chứng chỉ.
    function revokeCertificate(bytes32 certId) external {
        Certificate storage cert = certificates[certId];
        require(cert.status == Status.Issued, "CredentialRegistry: certificate not in Issued state");

        bool byAdmin = msg.sender == issuerAdmin;
        bool byHotKey = issuerStatus == IssuerStatus.Active &&
            ((msg.sender == cert.issuer && isIssuer[cert.issuer]) || 
            msg.sender == issuerSuccessor[cert.issuer]);
        require(byAdmin || byHotKey, "CredentialRegistry: not authorized to revoke");

        cert.status = Status.Revoked;
        cert.revokedAt = uint64(block.timestamp);
        emit CertificateRevoked(certId, msg.sender, block.timestamp);
    }

    // ---------- Thu hồi cứu hộ có thời gian chờ ----------

    /// @notice Owner ĐỀ XUẤT thu hồi. Không thi hành ngay: issuer còn sống có
    ///         `gracePeriod` để phủ quyết. Im lặng là đồng ý.
    function proposeOwnerRevoke(bytes32 certId, string calldata reason) external onlyOwner {
        require(certificates[certId].status == Status.Issued,
                "CredentialRegistry: certificate not in Issued state");
        require(pendingRevoke[certId] == 0, "CredentialRegistry: already proposed");
        uint64 executeAfter = uint64(block.timestamp) + gracePeriod;
        pendingRevoke[certId] = executeAfter;
        emit OwnerRevokeProposed(certId, reason, executeAfter);
    }

    /// @notice Chống owner lạm quyền.
    function cancelOwnerRevoke(bytes32 certId) external onlyIssuerAdmin {
        require(pendingRevoke[certId] != 0, "CredentialRegistry: nothing proposed");
        delete pendingRevoke[certId];
        emit OwnerRevokeCancelled(certId, msg.sender);
    }

    /// @notice Đề xuất đã được đưa ra mà không bị phủ quyết thì
    ///         bất kỳ ai cũng thi hành được — một học viên, một nhà báo, một kiểm toán viên.
    ///         Nhờ vậy việc cứu hộ không chết theo đơn vị vận hành.
    function executeOwnerRevoke(bytes32 certId) external {
        uint64 t = pendingRevoke[certId];
        require(t != 0, "CredentialRegistry: nothing proposed");
        require(block.timestamp >= t, "CredentialRegistry: still in grace period");

        Certificate storage cert = certificates[certId];
        require(cert.status == Status.Issued, "CredentialRegistry: certificate not in Issued state");

        delete pendingRevoke[certId];
        cert.status = Status.Revoked;
        cert.revokedAt = uint64(block.timestamp);
        emit CertificateRevoked(certId, msg.sender, block.timestamp);
    }

    // ---------- Chuyển quyền ----------

    function transferOwnership(address newOwner) external onlyOwner {
        require(newOwner != address(0), "CredentialRegistry: zero address");
        pendingOwner = newOwner;
        emit OwnershipTransferStarted(owner, newOwner);
    }

    /// @dev Hai bước: gõ nhầm địa chỉ không làm mất quyền vĩnh viễn.
    function acceptOwnership() external {
        require(msg.sender == pendingOwner, "CredentialRegistry: not pending owner");
        emit OwnershipTransferred(owner, pendingOwner);
        owner = pendingOwner;
        pendingOwner = address(0);
    }

    function transferIssuerAdmin(address newAdmin) external onlyIssuerAdmin {
        require(newAdmin != address(0), "CredentialRegistry: zero address");
        pendingIssuerAdmin = newAdmin;
        emit IssuerAdminTransferStarted(issuerAdmin, newAdmin);
    }

    function acceptIssuerAdmin() external {
        require(msg.sender == pendingIssuerAdmin, "CredentialRegistry: not pending issuer admin");
        emit IssuerAdminTransferred(issuerAdmin, pendingIssuerAdmin);
        issuerAdmin = pendingIssuerAdmin;
        pendingIssuerAdmin = address(0);
    }

    // ---------- Quyền thoát ly ----------

    /// @notice Đơn vị phát hành giành quyền tự chủ. Owner KHÔNG hủy được.
    function requestSelfSovereignty() external onlyIssuerAdmin {
        exitRequestedAt = uint64(block.timestamp);
        emit SelfSovereigntyRequested(uint256(exitRequestedAt) + exitDelay);
    }

    /// @notice Đổi ý thì rút lại được — nhưng chỉ chính issuerAdmin mới rút.
    function cancelSelfSovereignty() external onlyIssuerAdmin {
        require(exitRequestedAt != 0, "CredentialRegistry: no exit requested");
        exitRequestedAt = 0;
        emit SelfSovereigntyCancelled();
    }

    function executeSelfSovereignty() external onlyIssuerAdmin {
        require(exitRequestedAt != 0, "CredentialRegistry: no exit requested");
        require(block.timestamp >= uint256(exitRequestedAt) + exitDelay,
                "CredentialRegistry: exit delay not elapsed");
        exitRequestedAt = 0;
        pendingOwner = address(0);
        emit OwnershipTransferred(owner, issuerAdmin);
        owner = issuerAdmin;
        emit SelfSovereigntyExecuted(owner);
    }

    // ---------- Đọc ----------

    /// @notice Xác minh một chứng chỉ. Chỉ đọc, không tốn gas, KHÔNG BAO GIỜ bị chặn kể cả khi đang tranh chấp.
    function verifyCertificate(bytes32 certId, bytes32 providedHash)
        external
        view
        returns (bool valid, Status status, address issuer, address holder, uint256 issuedAt)
    {
        Certificate storage cert = certificates[certId];
        bool hashMatches  = cert.certHash == providedHash;
        bool isValidStatus = cert.status == Status.Issued;
        valid = hashMatches && isValidStatus;
        return (valid, cert.status, cert.issuer, cert.holder, uint256(cert.issuedAt));
    }

    /// @notice Trả về toàn bộ bản ghi dưới dạng struct có tên trường.
    function getCertificate(bytes32 certId) external view returns (Certificate memory) {
        return certificates[certId];
    }

    /// @notice Liệt kê những khóa ĐANG có quyền cấp.
    function listActiveIssuers() external view returns (address[] memory active) {
        uint256 n = 0;
        for (uint256 i = 0; i < _knownIssuers.length; i++) {
            if (isIssuer[_knownIssuers[i]]) n++;
        }
        active = new address[](n);
        uint256 j = 0;
        for (uint256 i = 0; i < _knownIssuers.length; i++) {
            if (isIssuer[_knownIssuers[i]]) active[j++] = _knownIssuers[i];
        }
    }

    /// @notice Mọi địa chỉ TỪNG được cấp quyền, kể cả đã bị gỡ — phục vụ kiểm toán.
    function knownIssuers() external view returns (address[] memory) {
        return _knownIssuers;
    }

    /// @notice Gói toàn bộ trạng thái quản trị vào một lời gọi.
    function governance()
        external
        view
        returns (
            address owner_,
            address pendingOwner_,
            address issuerAdmin_,
            address pendingIssuerAdmin_,
            IssuerStatus status_,
            uint64 gracePeriod_,
            uint64 exitDelay_,
            uint64 exitRequestedAt_,
            uint256 activeIssuerCount_
        )
    {
        return (
            owner, pendingOwner, issuerAdmin, pendingIssuerAdmin,
            issuerStatus, gracePeriod, exitDelay, exitRequestedAt, activeIssuerCount
        );
    }
}
