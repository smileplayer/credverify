const { expect } = require("chai");
const { ethers } = require("hardhat");
const { time } = require("@nomicfoundation/hardhat-network-helpers");
const { anyValue } = require("@nomicfoundation/hardhat-chai-matchers/withArgs");

const GRACE = 7 * 24 * 60 * 60;   // 7 ngày
const EXIT = 30 * 24 * 60 * 60;   // 30 ngày

describe("CredentialRegistry", function () {
  let registry;
  let deployer, issuerAdmin, trainingCenter, backupKey, otherIssuer, student, attacker;
  let certId, certHash;

  beforeEach(async function () {
    [deployer, issuerAdmin, trainingCenter, backupKey, otherIssuer, student, attacker] =
      await ethers.getSigners();

    const CredentialRegistry = await ethers.getContractFactory("CredentialRegistry");
    registry = await CredentialRegistry.deploy(
      issuerAdmin.address,      // khóa lạnh
      trainingCenter.address,   // khóa nóng đầu tiên
      GRACE,
      EXIT
    );
    await registry.waitForDeployment();

    certId = ethers.keccak256(ethers.toUtf8Bytes("KHOAHOC-2026-0001"));
    certHash = ethers.keccak256(ethers.toUtf8Bytes("noi-dung-file-pdf-chung-chi"));
  });

  const statusOf = async (id) => Number((await registry.getCertificate(id)).status);

  // =========================================================================
  // HAPPY PATH
  // =========================================================================

  it("issuer hợp lệ cấp chứng chỉ thành công và phát event CertificateIssued", async function () {
    await expect(
      registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address)
    )
      .to.emit(registry, "CertificateIssued")
      .withArgs(certId, certHash, trainingCenter.address, student.address, anyValue);

    const [valid, status] = await registry.verifyCertificate(certId, certHash);
    expect(valid).to.equal(true);
    expect(status).to.equal(1);
  });

  it("verifier xác minh đúng hash trả về valid = true", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    const [valid] = await registry.verifyCertificate(certId, certHash);
    expect(valid).to.equal(true);
  });

  it("verifier xác minh với hash sai (tệp bị chỉnh sửa) trả về valid = false", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    const wrongHash = ethers.keccak256(ethers.toUtf8Bytes("file-da-bi-sua"));
    const [valid] = await registry.verifyCertificate(certId, wrongHash);
    expect(valid).to.equal(false);
  });

  it("issuer đúng người thu hồi chứng chỉ thành công và ghi lại revokedAt", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);

    await expect(registry.connect(trainingCenter).revokeCertificate(certId))
      .to.emit(registry, "CertificateRevoked");

    const [valid, status] = await registry.verifyCertificate(certId, certHash);
    expect(valid).to.equal(false);
    expect(status).to.equal(2);

    // Thời điểm thu hồi được lưu on-chain, phục vụ câu hỏi
    // "lúc tuyển dụng, tấm bằng này còn hiệu lực không?"
    const rec = await registry.getCertificate(certId);
    expect(rec.revokedAt).to.be.greaterThan(0);
    expect(rec.revokedAt).to.be.greaterThanOrEqual(rec.issuedAt);
  });

  it("issuerAdmin thêm được khóa cấp mới và khóa đó cấp được chứng chỉ", async function () {
    await registry.connect(issuerAdmin).addIssuer(otherIssuer.address);
    await expect(
      registry.connect(otherIssuer).issueCertificate(certId, certHash, student.address)
    ).to.emit(registry, "CertificateIssued");
  });

  // =========================================================================
  // HÀNH VI SAI BỊ CHẶN
  // =========================================================================

  it("BỊ CHẶN: địa chỉ không phải issuer cố cấp chứng chỉ => revert", async function () {
    await expect(
      registry.connect(attacker).issueCertificate(certId, certHash, student.address)
    ).to.be.revertedWith("CredentialRegistry: caller is not an authorized issuer");
  });

  it("BỊ CHẶN: khóa cấp khác (không phải khóa đã cấp) cố thu hồi => revert", async function () {
    await registry.connect(issuerAdmin).addIssuer(otherIssuer.address);
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);

    await expect(
      registry.connect(otherIssuer).revokeCertificate(certId)
    ).to.be.revertedWith("CredentialRegistry: not authorized to revoke");
  });

  it("BỊ CHẶN: cấp trùng certId đã tồn tại => revert", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await expect(
      registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address)
    ).to.be.revertedWith("CredentialRegistry: certId already used");
  });

  it("BỊ CHẶN: thu hồi hai lần trên cùng một chứng chỉ => revert", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(trainingCenter).revokeCertificate(certId);
    await expect(
      registry.connect(trainingCenter).revokeCertificate(certId)
    ).to.be.revertedWith("CredentialRegistry: certificate not in Issued state");
  });

  it("BỊ CHẶN: owner KHÔNG tự thêm được khóa cấp — đơn vị vận hành không được phát hành nhân danh khách hàng", async function () {
    await expect(
      registry.connect(deployer).addIssuer(deployer.address)
    ).to.be.revertedWith("CredentialRegistry: caller is not issuer admin");
  });

  it("BỊ CHẶN: người lạ cố thêm khóa cấp => revert", async function () {
    await expect(
      registry.connect(attacker).addIssuer(attacker.address)
    ).to.be.revertedWith("CredentialRegistry: caller is not issuer admin");
  });

  it("BỊ CHẶN: xác minh chứng chỉ không tồn tại => valid = false, status = None", async function () {
    const unknownId = ethers.keccak256(ethers.toUtf8Bytes("KHOAHOC-2026-UNKNOWN"));
    const unknownHash = ethers.keccak256(ethers.toUtf8Bytes("file-khong-ton-tai"));

    const [valid, status, issuer, holder, issuedAt] =
      await registry.verifyCertificate(unknownId, unknownHash);

    expect(valid).to.equal(false);
    expect(status).to.equal(0);
    expect(issuer).to.equal(ethers.ZeroAddress);
    expect(holder).to.equal(ethers.ZeroAddress);
    expect(issuedAt).to.equal(0);
  });

  it("BỊ CHẶN: thu hồi chứng chỉ không tồn tại => revert", async function () {
    const unknownId = ethers.keccak256(ethers.toUtf8Bytes("KHOAHOC-2026-UNKNOWN"));
    await expect(
      registry.connect(trainingCenter).revokeCertificate(unknownId)
    ).to.be.revertedWith("CredentialRegistry: certificate not in Issued state");
  });

  it("BỊ CHẶN: cấp chứng chỉ với holder là zero address => revert", async function () {
    await expect(
      registry.connect(trainingCenter).issueCertificate(certId, certHash, ethers.ZeroAddress)
    ).to.be.revertedWith("CredentialRegistry: holder is zero address");
  });

  it("BỊ CHẶN: cấp chứng chỉ với certHash rỗng => revert", async function () {
    await expect(
      registry.connect(trainingCenter).issueCertificate(certId, ethers.ZeroHash, student.address)
    ).to.be.revertedWith("CredentialRegistry: empty certHash");
  });

  it("BỊ CHẶN: issuerAdmin thêm khóa cấp là zero address => revert", async function () {
    await expect(
      registry.connect(issuerAdmin).addIssuer(ethers.ZeroAddress)
    ).to.be.revertedWith("CredentialRegistry: zero address");
  });

  it("BỊ CHẶN: gỡ quyền một địa chỉ chưa từng là issuer => revert (giữ sạch audit trail)", async function () {
    await expect(
      registry.connect(issuerAdmin).removeIssuer(attacker.address)
    ).to.be.revertedWith("CredentialRegistry: not an issuer");
  });

  it("BỊ CHẶN: constructor với exitDelay <= gracePeriod => revert", async function () {
    const F = await ethers.getContractFactory("CredentialRegistry");
    await expect(
      F.deploy(issuerAdmin.address, trainingCenter.address, GRACE, GRACE)
    ).to.be.revertedWith("CredentialRegistry: exitDelay must exceed gracePeriod");
  });

  it("BỊ CHẶN: certificate đã revoke không thể được issue lại bằng cùng certId", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(trainingCenter).revokeCertificate(certId);
    await expect(
      registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address)
    ).to.be.revertedWith("CredentialRegistry: certId already used");
  });

  // =========================================================================
  // BẤT BIẾN NGHIỆP VỤ
  // =========================================================================

  it("certificate giữ nguyên issuer và holder sau khi bị thu hồi", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(trainingCenter).revokeCertificate(certId);

    const rec = await registry.getCertificate(certId);
    expect(rec.issuer).to.equal(trainingCenter.address);
    expect(rec.holder).to.equal(student.address);
    expect(rec.issuedAt).to.be.greaterThan(0);
  });

  it("hai certId khác nhau tạo hai certificate độc lập", async function () {
    const certId2 = ethers.keccak256(ethers.toUtf8Bytes("KHOAHOC-2026-0002"));
    const certHash2 = ethers.keccak256(ethers.toUtf8Bytes("noi-dung-2"));

    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(trainingCenter).issueCertificate(certId2, certHash2, student.address);

    expect((await registry.verifyCertificate(certId, certHash))[0]).to.equal(true);
    expect((await registry.verifyCertificate(certId2, certHash2))[0]).to.equal(true);
  });

  it("chứng chỉ đã cấp vẫn xác minh được sau khi khóa cấp bị gỡ quyền", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(issuerAdmin).removeIssuer(trainingCenter.address);

    const [valid, status] = await registry.verifyCertificate(certId, certHash);
    expect(valid).to.equal(true);
    expect(status).to.equal(1);
  });

  // =========================================================================
  // QUYẾT ĐỊNH THIẾT KẾ CÓ CHỦ ĐÍCH
  // =========================================================================

  it("THIẾT KẾ ĐÃ ĐẢO NGƯỢC: khóa bị gỡ quyền KHÔNG còn thu hồi được nữa", async function () {
    // Phiên bản trước cho phép khóa đã bị gỡ quyền vẫn thu hồi chứng chỉ của mình,
    // với lập luận "trung tâm ngừng hợp tác vẫn cần dọn dẹp được".
    //
    // Khi phân tích kịch bản lộ khóa, quyết định đó lộ ra một lỗ hổng: kẻ tấn công giữ
    // bản sao khóa đã lộ có thể thu hồi TOÀN BỘ chứng chỉ hợp lệ của trung tâm, vĩnh viễn,
    // và removeIssuer không cắt được.
    //
    // Thiết kế được sửa: removeIssuer nay cắt TOÀN BỘ quyền lực của một khóa.
    // Quyền thu hồi lịch sử chuyển cho khóa kế nhiệm hoặc issuerAdmin.
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(issuerAdmin).removeIssuer(trainingCenter.address);

    await expect(
      registry.connect(trainingCenter).revokeCertificate(certId)
    ).to.be.revertedWith("CredentialRegistry: not authorized to revoke");

    // Nhưng chứng chỉ KHÔNG bị mắc kẹt: issuerAdmin vẫn thu hồi được.
    await expect(registry.connect(issuerAdmin).revokeCertificate(certId))
      .to.emit(registry, "CertificateRevoked");
  });

  it("THIẾT KẾ CÓ CHỦ ĐÍCH: verifyCertificate không bao giờ bị chặn, kể cả khi đang tranh chấp", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(deployer).setDisputed("nghi ngo lo khoa");

    const [valid, status] = await registry.verifyCertificate(certId, certHash);
    expect(valid).to.equal(true);
    expect(status).to.equal(1);
  });

  // =========================================================================
  // XOAY KHÓA
  // =========================================================================

  it("XOAY KHÓA: khóa kế nhiệm thu hồi được chứng chỉ do khóa cũ đã cấp", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);

    // Nghi lộ khóa: gỡ quyền khóa cũ, chỉ định khóa kế nhiệm
    await registry.connect(issuerAdmin).removeIssuer(trainingCenter.address);
    await registry.connect(issuerAdmin).addIssuer(backupKey.address);
    await expect(registry.connect(issuerAdmin).setIssuerSuccessor(trainingCenter.address, backupKey.address))
      .to.emit(registry, "IssuerSuccessorSet")
      .withArgs(trainingCenter.address, backupKey.address);

    await expect(registry.connect(backupKey).revokeCertificate(certId))
      .to.emit(registry, "CertificateRevoked");
    expect(await statusOf(certId)).to.equal(2);
  });

  it("BỊ CHẶN: chỉ issuerAdmin chỉ định được khóa kế nhiệm", async function () {
    await expect(
      registry.connect(deployer).setIssuerSuccessor(trainingCenter.address, attacker.address)
    ).to.be.revertedWith("CredentialRegistry: caller is not issuer admin");
  });

  // =========================================================================
  // TRẠNG THÁI TRANH CHẤP
  // =========================================================================

  it("TRANH CHẤP: khóa nóng bị vô hiệu hóa ngay lập tức", async function () {
    await expect(registry.connect(deployer).setDisputed("nghi ngo lo khoa"))
      .to.emit(registry, "IssuerDisputed");
    expect(await registry.issuerStatus()).to.equal(1);

    await expect(
      registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address)
    ).to.be.revertedWith("CredentialRegistry: issuer is disputed");
  });

  it("TRANH CHẤP: khóa nóng KHÔNG thu hồi được, nhưng issuerAdmin thì luôn được — phanh không bị khóa", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(deployer).setDisputed("nghi ngo lo khoa");

    await expect(
      registry.connect(trainingCenter).revokeCertificate(certId)
    ).to.be.revertedWith("CredentialRegistry: not authorized to revoke");

    await expect(registry.connect(issuerAdmin).revokeCertificate(certId))
      .to.emit(registry, "CertificateRevoked");
  });

  it("TRANH CHẤP: issuerAdmin gỡ được TỨC THÌ — lá chắn chống owner dùng setDisputed làm vũ khí", async function () {
    await registry.connect(deployer).setDisputed("chan dich vu");
    await expect(registry.connect(issuerAdmin).restore()).to.emit(registry, "IssuerRestored");
    expect(await registry.issuerStatus()).to.equal(0);

    await expect(
      registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address)
    ).to.emit(registry, "CertificateIssued");
  });

  it("BỊ CHẶN: người lạ không đặt được trạng thái tranh chấp", async function () {
    await expect(
      registry.connect(attacker).setDisputed("pha hoai")
    ).to.be.revertedWith("CredentialRegistry: caller is not owner or issuer admin");
  });

  // =========================================================================
  // THU HỒI CỨU HỘ CÓ THỜI GIAN CHỜ
  // =========================================================================

  it("CỨU HỘ: owner đề xuất, hết thời gian chờ thì thi hành được", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);

    await expect(registry.connect(deployer).proposeOwnerRevoke(certId, "trung tam da giai the"))
      .to.emit(registry, "OwnerRevokeProposed");

    await time.increase(GRACE + 1);
    await expect(registry.connect(deployer).executeOwnerRevoke(certId))
      .to.emit(registry, "CertificateRevoked");
    expect(await statusOf(certId)).to.equal(2);
  });

  it("CỨU HỘ: thi hành trước khi hết thời gian chờ => revert", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(deployer).proposeOwnerRevoke(certId, "ly do");

    await time.increase(GRACE - 3600);
    await expect(
      registry.connect(deployer).executeOwnerRevoke(certId)
    ).to.be.revertedWith("CredentialRegistry: still in grace period");
  });

  it("PHỦ QUYẾT: issuerAdmin hủy được đề xuất của owner", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(deployer).proposeOwnerRevoke(certId, "lam quyen");

    await expect(registry.connect(issuerAdmin).cancelOwnerRevoke(certId))
      .to.emit(registry, "OwnerRevokeCancelled");

    await time.increase(GRACE + 1);
    await expect(
      registry.connect(deployer).executeOwnerRevoke(certId)
    ).to.be.revertedWith("CredentialRegistry: nothing proposed");
    expect(await statusOf(certId)).to.equal(1);   // vẫn còn hiệu lực
  });

  it("BỊ CHẶN: owner KHÔNG tự hủy được đề xuất của chính mình (chỉ issuerAdmin phủ quyết)", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(deployer).proposeOwnerRevoke(certId, "ly do");
    await expect(
      registry.connect(deployer).cancelOwnerRevoke(certId)
    ).to.be.revertedWith("CredentialRegistry: caller is not issuer admin");
  });

  it("NGUYÊN TẮC 4: đề xuất đã chín thì BẤT KỲ AI cũng thi hành được", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(deployer).proposeOwnerRevoke(certId, "trung tam da giai the");
    await time.increase(GRACE + 1);

    // student — một người ngoài hoàn toàn — thi hành được.
    // Nhờ vậy việc cứu hộ không chết theo đơn vị vận hành.
    await expect(registry.connect(student).executeOwnerRevoke(certId))
      .to.emit(registry, "CertificateRevoked");
  });

  it("BỊ CHẶN: người không phải owner không đề xuất được thu hồi cứu hộ", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await expect(
      registry.connect(attacker).proposeOwnerRevoke(certId, "pha hoai")
    ).to.be.revertedWith("CredentialRegistry: caller is not owner");
  });

  // =========================================================================
  // CHUYỂN QUYỀN
  // =========================================================================

  it("CHUYỂN QUYỀN: transferOwnership hai bước", async function () {
    await registry.connect(deployer).transferOwnership(otherIssuer.address);
    expect(await registry.owner()).to.equal(deployer.address);   // chưa đổi

    await expect(registry.connect(otherIssuer).acceptOwnership())
      .to.emit(registry, "OwnershipTransferred");
    expect(await registry.owner()).to.equal(otherIssuer.address);
  });

  it("BỊ CHẶN: người không được chỉ định không nhận được quyền owner", async function () {
    await registry.connect(deployer).transferOwnership(otherIssuer.address);
    await expect(
      registry.connect(attacker).acceptOwnership()
    ).to.be.revertedWith("CredentialRegistry: not pending owner");
  });

  it("CHUYỂN QUYỀN: transferIssuerAdmin hai bước", async function () {
    await registry.connect(issuerAdmin).transferIssuerAdmin(backupKey.address);
    expect(await registry.issuerAdmin()).to.equal(issuerAdmin.address);

    await expect(registry.connect(backupKey).acceptIssuerAdmin())
      .to.emit(registry, "IssuerAdminTransferred");
    expect(await registry.issuerAdmin()).to.equal(backupKey.address);
  });

  // =========================================================================
  // QUYỀN THOÁT LY
  // =========================================================================

  it("THOÁT LY: issuerAdmin giành được quyền owner sau thời gian chờ, owner không cản được", async function () {
    await expect(registry.connect(issuerAdmin).requestSelfSovereignty())
      .to.emit(registry, "SelfSovereigntyRequested");

    await time.increase(EXIT + 1);
    await expect(registry.connect(issuerAdmin).executeSelfSovereignty())
      .to.emit(registry, "SelfSovereigntyExecuted");

    expect(await registry.owner()).to.equal(issuerAdmin.address);
  });

  it("THOÁT LY: thi hành trước hạn => revert", async function () {
    await registry.connect(issuerAdmin).requestSelfSovereignty();
    await time.increase(EXIT - 3600);
    await expect(
      registry.connect(issuerAdmin).executeSelfSovereignty()
    ).to.be.revertedWith("CredentialRegistry: exit delay not elapsed");
  });

  it("THOÁT LY: chỉ issuerAdmin yêu cầu được, owner không hủy được", async function () {
    await expect(
      registry.connect(attacker).requestSelfSovereignty()
    ).to.be.revertedWith("CredentialRegistry: caller is not issuer admin");

    await registry.connect(issuerAdmin).requestSelfSovereignty();
    await expect(
      registry.connect(deployer).cancelSelfSovereignty()
    ).to.be.revertedWith("CredentialRegistry: caller is not issuer admin");
  });

  it("THOÁT LY: không dùng được để chuồn khỏi một đề xuất thu hồi đang chờ", async function () {
    // exitDelay (30 ngày) > gracePeriod (7 ngày), và executeOwnerRevoke không cần onlyOwner.
    // Nên đề xuất luôn hạ cánh trước khi việc thoát ly có hiệu lực.
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(deployer).proposeOwnerRevoke(certId, "cap bang gia");
    await registry.connect(issuerAdmin).requestSelfSovereignty();

    await time.increase(GRACE + 1);
    await registry.connect(student).executeOwnerRevoke(certId);   // người ngoài thi hành
    expect(await statusOf(certId)).to.equal(2);

    // Thoát ly vẫn diễn ra, nhưng chứng chỉ giả đã bị vô hiệu hóa trước đó.
    await time.increase(EXIT + 1);
    await registry.connect(issuerAdmin).executeSelfSovereignty();
    expect(await registry.owner()).to.equal(issuerAdmin.address);
    expect(await statusOf(certId)).to.equal(2);
  });

  // =========================================================================
  // KỊCH BẢN ĐẦU–CUỐI: LỘ KHÓA CẤP
  // =========================================================================

  it("KỊCH BẢN LỘ KHÓA: phát hiện -> đóng băng -> xoay khóa -> dọn hậu quả -> khôi phục", async function () {
    // 1. Hoạt động bình thường
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);

    // 2. Kẻ tấn công chiếm khóa nóng và cấp chứng chỉ giả
    const fakeId = ethers.keccak256(ethers.toUtf8Bytes("BANG-GIA-001"));
    const fakeHash = ethers.keccak256(ethers.toUtf8Bytes("noi-dung-bang-gia"));
    await registry.connect(trainingCenter).issueCertificate(fakeId, fakeHash, attacker.address);
    expect(await statusOf(fakeId)).to.equal(1);

    // 3. Phát hiện: đóng băng tức thì, không cần chờ
    await registry.connect(deployer).setDisputed("khoa cap bi lo");
    const fakeId2 = ethers.keccak256(ethers.toUtf8Bytes("BANG-GIA-002"));
    await expect(
      registry.connect(trainingCenter).issueCertificate(fakeId2, fakeHash, attacker.address)
    ).to.be.revertedWith("CredentialRegistry: issuer is disputed");

    // 4. Kẻ tấn công KHÔNG dùng khóa đã lộ để phá hoại được nữa
    await expect(
      registry.connect(trainingCenter).revokeCertificate(certId)
    ).to.be.revertedWith("CredentialRegistry: not authorized to revoke");

    // 5. Dọn hậu quả bằng khóa lạnh
    await registry.connect(issuerAdmin).revokeCertificate(fakeId);
    expect(await statusOf(fakeId)).to.equal(2);

    // 6. Xoay khóa và khôi phục hoạt động
    await registry.connect(issuerAdmin).removeIssuer(trainingCenter.address);
    await registry.connect(issuerAdmin).addIssuer(backupKey.address);
    await registry.connect(issuerAdmin).setIssuerSuccessor(trainingCenter.address, backupKey.address);
    await registry.connect(issuerAdmin).restore();

    const newId = ethers.keccak256(ethers.toUtf8Bytes("KHOAHOC-2026-0100"));
    await expect(
      registry.connect(backupKey).issueCertificate(newId, certHash, student.address)
    ).to.emit(registry, "CertificateIssued");

    // 7. Chứng chỉ hợp lệ của học viên không hề bị ảnh hưởng suốt quá trình
    expect((await registry.verifyCertificate(certId, certHash))[0]).to.equal(true);

    // 8. Khóa đã lộ vô hại vĩnh viễn
    await expect(
      registry.connect(trainingCenter).revokeCertificate(certId)
    ).to.be.revertedWith("CredentialRegistry: not authorized to revoke");
  });

  // =========================================================================
  // TRUY VẤN QUA EVENT
  // =========================================================================

  it("Học viên truy vấn được đúng danh sách chứng chỉ của mình qua event", async function () {
    const certId2 = ethers.keccak256(ethers.toUtf8Bytes("KHOAHOC-2026-0002"));
    const certHash2 = ethers.keccak256(ethers.toUtf8Bytes("noi-dung-2"));

    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(trainingCenter).issueCertificate(certId2, certHash2, student.address);

    const logs = await registry.queryFilter(
      registry.filters.CertificateIssued(null, null, null, student.address), 0, "latest");

    expect(logs.length).to.equal(2);
    const ids = logs.map((l) => l.args.certId);
    expect(ids).to.include(certId);
    expect(ids).to.include(certId2);
  });

  it("Danh sách của học viên không lẫn chứng chỉ của ví khác", async function () {
    const otherId = ethers.keccak256(ethers.toUtf8Bytes("KHOAHOC-2026-0009"));

    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    await registry.connect(trainingCenter).issueCertificate(otherId, certHash, attacker.address);

    const mine = await registry.queryFilter(
      registry.filters.CertificateIssued(null, null, null, student.address), 0, "latest");
    expect(mine.length).to.equal(1);
    expect(mine[0].args.certId).to.equal(certId);
  });

  it("Xác minh không cần mã: tra ngược được certId từ hash của tệp", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);

    const logs = await registry.queryFilter(
      registry.filters.CertificateIssued(null, certHash, null, null), 0, "latest");

    expect(logs.length).to.equal(1);
    expect(logs[0].args.certId).to.equal(certId);
    expect((await registry.verifyCertificate(logs[0].args.certId, certHash))[0]).to.equal(true);
  });

  it("Xác minh không cần mã: tệp chưa từng đăng ký thì không tra ra bản ghi nào", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    const strangeHash = ethers.keccak256(ethers.toUtf8Bytes("tep-chua-tung-duoc-cap"));
    const logs = await registry.queryFilter(
      registry.filters.CertificateIssued(null, strangeHash, null, null), 0, "latest");
    expect(logs.length).to.equal(0);
  });

  // =========================================================================
  // ĐỌC TRẠNG THÁI QUẢN TRỊ
  // =========================================================================

  // =========================================================================
  // KHẢ NĂNG NHÌN THẤY TẬP KHÓA CẤP
  // =========================================================================

  it("KHẢ KIẾN: đếm và liệt kê được các khóa cấp đang có quyền", async function () {
    expect(await registry.activeIssuerCount()).to.equal(1);
    expect([...(await registry.listActiveIssuers())]).to.deep.equal([trainingCenter.address]);

    await registry.connect(issuerAdmin).addIssuer(backupKey.address);
    expect(await registry.activeIssuerCount()).to.equal(2);
    expect([...(await registry.listActiveIssuers())]).to.have.members([trainingCenter.address, backupKey.address]);

    await registry.connect(issuerAdmin).removeIssuer(trainingCenter.address);
    expect(await registry.activeIssuerCount()).to.equal(1);
    expect([...(await registry.listActiveIssuers())]).to.deep.equal([backupKey.address]);
  });

  it("KHẢ KIẾN: một khóa được thêm lén KHÔNG còn vô hình", async function () {
    const before = await registry.activeIssuerCount();
    await registry.connect(issuerAdmin).addIssuer(attacker.address);
    const after = await registry.activeIssuerCount();

    expect(after).to.equal(before + 1n);
    expect([...(await registry.listActiveIssuers())]).to.include(attacker.address);

    await expect(
      registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address)
    ).to.emit(registry, "CertificateIssued");
  });

  it("KHẢ KIẾN: knownIssuers() giữ cả khóa đã bị gỡ, phục vụ kiểm toán", async function () {
    await registry.connect(issuerAdmin).addIssuer(backupKey.address);
    await registry.connect(issuerAdmin).removeIssuer(backupKey.address);

    expect([...(await registry.listActiveIssuers())]).to.not.include(backupKey.address);
    expect([...(await registry.knownIssuers())]).to.include(backupKey.address);
  });

  it("KHẢ KIẾN: cấp lại quyền cho một khóa cũ không làm trùng bản ghi", async function () {
    await registry.connect(issuerAdmin).removeIssuer(trainingCenter.address);
    await registry.connect(issuerAdmin).addIssuer(trainingCenter.address);

    expect(await registry.activeIssuerCount()).to.equal(1);
    const known = [...(await registry.knownIssuers())];
    expect(known.filter((a) => a === trainingCenter.address).length).to.equal(1);
  });

  it("governance() trả về toàn bộ trạng thái quản trị trong một lời gọi", async function () {
    const g = await registry.governance();
    expect(g.owner_).to.equal(deployer.address);
    expect(g.issuerAdmin_).to.equal(issuerAdmin.address);
    expect(g.status_).to.equal(0);
    expect(g.gracePeriod_).to.equal(GRACE);
    expect(g.exitDelay_).to.equal(EXIT);
    expect(g.exitRequestedAt_).to.equal(0);
    expect(g.activeIssuerCount_).to.equal(1);
  });

  it("getCertificate() đọc theo tên trường, không vỡ khi struct đổi thứ tự", async function () {
    await registry.connect(trainingCenter).issueCertificate(certId, certHash, student.address);
    const rec = await registry.getCertificate(certId);
    expect(rec.certHash).to.equal(certHash);
    expect(rec.issuer).to.equal(trainingCenter.address);
    expect(rec.holder).to.equal(student.address);
    expect(Number(rec.status)).to.equal(1);
    expect(rec.revokedAt).to.equal(0);
  });
